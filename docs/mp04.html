## 1. Request the CES total nonfarm series table (all years, all months)
resp <- request("https://data.bls.gov/pdq/SurveyOutputServlet") |>
  req_method("POST") |>
  req_body_form(
    survey    = "ce",
    series_id = "CES0000000001",  # total nonfarm, seasonally adjusted
    from_year = "1979",
    to_year   = "2025"
  ) |>
  req_perform()

## 2. Parse HTML from the response
ces_html <- resp |> resp_body_html()

## 3. Grab the main data table
ces_tables <- ces_html |>
  html_elements("table") |>
  html_table(fill = TRUE)

# Pick the table that has a 'Year' column
ces_raw <- ces_tables |>
  keep(~ "Year" %in% names(.x)) |>
  pluck(1)

## 4. Clean and reshape to date / level
ces_tidy <-
  ces_raw |>
  # keep only rows that are actual years (drop headers/footers)
  filter(str_detect(Year, "^[0-9]{4}$")) |>
  mutate(Year = as.integer(Year)) |>
  pivot_longer(
    cols      = -Year,
    names_to  = "month",
    values_to = "level"
  ) |>
  # keep only real months, drop "Annual" or other summary columns
  filter(month %in% month.abb) |>
  mutate(
    # remove footnote chars (like * or p), keep digits, dot, minus
    level = as.numeric(gsub("[^0-9.-]", "", level)),
    # build a proper Date from "1979 Jan" → 1979-01-01
    date  = ym(paste(Year, month))
  ) |>
  arrange(date) |>
  # restrict to Jan 1979 – Jun 2025
  filter(
    date >= ymd("1979-01-01"),
    date <= ymd("2025-06-01")
  ) |>
  select(date, level) |>
  drop_na()

resp_check_status(rev_resp)

rev_html <- resp_body_html(rev_resp)

## 2. Grab ALL tables on the page
rev_tables <- rev_html |> html_elements("table")

## 3. Helper: pull out the month/year + 1st & 3rd estimates from ONE table
extract_revision_rows <- function(tbl_node) {
  df <- tbl_node |>
    html_table(header = FALSE, fill = TRUE) |>
    as_tibble()

  # Give generic names V1, V2, ...
  names(df) <- paste0("V", seq_len(ncol(df)))

  # Need at least 5 columns: month, year, 1st, 2nd, 3rd
  if (ncol(df) < 5) return(NULL)

  # Work with character for month/year
  df <- df |>
    mutate(
      V1 = as.character(V1),
      V2 = as.character(V2)
    )

  # Match "Jan.", "Feb.", ..., with or without the dot
  month_pat <- "^(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\\.?$"

  # Keep rows that look like a real month + 4-digit year
  rows <- df |>
    filter(
      str_detect(V1, month_pat),
      str_detect(V2, "^[0-9]{4}$")
    )

  if (nrow(rows) == 0) return(NULL)

  rows |>
    transmute(
      year     = as.integer(V2),
      # strip trailing "." so lubridate::ym can handle it
      month    = str_remove(V1, "\\.$"),
      # Seasonally adjusted 1st and 3rd estimates = original & final
      original = as.numeric(gsub("[^0-9-]", "", V3)),
      final    = as.numeric(gsub("[^0-9-]", "", V5))
    )
}

## 4. Apply to all tables and stack them
revisions_raw <- rev_tables |>
  map(extract_revision_rows) |>
  list_rbind()

## 5. Build final dataset: date, original, final, revision
ces_revisions <- revisions_raw |>
  mutate(
    date     = ym(paste(year, month)),
    revision = final - original
  ) |>
  # Keep only complete numeric rows in desired range
  filter(
    !is.na(date),
    !is.na(original),
    !is.na(final),
    date >= ymd("1979-01-01"),
    date <= ymd("2025-06-01")
  ) |>
  arrange(date) |>
  select(date, original, final, revision)

Task 4: Statistical Inference

For this task I will be using t_test and prop_test functions to perform statistical tests about CES revisions. From the one sample t-test, on average, the CES estimate understates employment by around 12k jobs, and the final numbers are revised up. So there is evidence of a small, systematic upward revision over the full period. When looking at the two sample t-test pre and post 2020 we don’t see strong evidence that average revisions changed in a meaningful way after 2020.Lastly based on the prop-test there’s no strong evidence that negative revisions became more or less common after 2000.

Task 5: Fact Checks of Claims about BLS

Based on all of the data we have gathered and analyzed, I will now be fact checking two claims about the BLS. The first claim is by President Trump. In his post announcing Erika McEntarfer’s firing, Trump said the jobs report was rigged to make him look bad and pointed to large downward revisions.

Between Jan 1979–Jun 2025, there are 1110 months with non-missing revisions. The average revision is about 12 thousand jobs, meaning the final estimate is slightly higher than the initial one, not lower. The average absolute revision is roughly 57 thousand jobs, which is only about 0.05% of the total employment level. As shown in Figure 2, revisions bounce above and below zero throughout the entire sample rather than being mostly negative in recent years. Figure 3 similarly shows a mix of positive and negative revisions in every month of the year.

We also did some hypothesis testing to see if the claims are supported or denied.

  1. Estimated mean revision: 12 thousand jobs
  2. 95% CI: [7.1, 16.9]
  3. p-value: 1.73^{-6}

Because the confidence interval is entirely positive and the p-value is extremely small I am going to reject this claim. Instead, the data suggest that revisions are on average upward, which contradicts the idea that BLS is rigging the numbers to make conditions look worse.Therefore we will be rating this claim as Pants on Fire.

The second claim we will be looking into would be by Bobby Scott from Virginia. After McEntarfer’s firing, Scott said that The Bureau of Labor Statistics has always served as a nonpartisan, data-driven federal agency that provides key information vital to the United States economy. Between January 1979 through June 2025, there are 1110 months with non-missing revision data. The average absolute revision is about 57 thousand jobs, while the typical employment level is on the ord. On average, revisions are therefore only about 0.05% of the employment level, far below the error rate some commentators have claimed.

Figure 4 shows the absolute CES revision as a percentage of the employment level over time. There are a few spikes around major shocks like recessions and COVID, but the baseline stays quite low and there is no long stretch where revisions are consistently huge. Taken together with Figure 1, which shows a steadily growing labor market with occasional downturns, this suggests that revision behavior is pretty stable relative to the scale of the economy.

To check whether recent years looked off or not, I compared the average revision pre-2020 vs post-2020 using a two-sample t-test. The estimated difference in mean revision is -8.6 thousand jobs, with a 95% confidence interval of [-34.3, 17.1] thousand jobs and a p-value of 0.508. Because the confidence interval includes zero and the p-value is large, I reject that average revisions in the 2020s are systematically different from the earlier period. Therefore I will rate the claim as Mostly True.
